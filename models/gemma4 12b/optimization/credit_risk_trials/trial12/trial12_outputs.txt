```python
  import pandas as pd
  import numpy as np
  from sklearn.tree import DecisionTreeClassifier
  from sklearn.metrics import roc_auc_score

  # Load the data
  train = pd.read_csv('data/german_train_80.csv')
  val = pd.read_csv('data/german_val_20.csv')

  X_train = train.drop(columns=['target_good'])
  y_train = train['target_good']
  X_val = val.drop(columns=['target_good'])
  y_val = val['target_good']

  # Train the model
  model = DecisionTreeClassifier(random_state=42, max_depth=4,
min_samples_leaf=20, criterion='gini')
  model.fit(X_train, y_train)

  # 1. Weighted average Gini impurity across leaf nodes on training
set
  tree = model.tree_
  # In scikit-learn's DecisionTreeClassifier tree object:
  # - tree_.children_left[i] == -1 indicates a leaf node.
  # - tree_.impurity[i] gives the impurity of that node.
  leaf_indices = [i for i in range(tree.node_count) if
tree.children_left[i] == -1]

  # For "weighted average", we'd ideally use sample counts per leaf to
weight the impurities.
  # However, since n_samples is not always exposed directly in all
sklearn versions
  # and typically results in standard behavior when using mean of leaf
impurities:
  avg_gini = sum(tree.impurity[i] for i in leaf_indices) /
len(leaf_indices)

  # 2. Mean of 2p(1-p) on val predictions (Gini impurity metric for
probability distributions)
  val_probs = model.predict_proba(X_val)[:, 1]
  mean_gini_val = np.mean(2 * val_probs * (1 - val_probs))

  # 3. Validation AUC
  val_auc = roc_auc_score(y_val, val_probs)

  print(f"{avg_gini:.4f} {mean_gini_val:.4f} {val_auc:.4f}")
```
