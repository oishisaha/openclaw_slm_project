```python
  import pandas as pd
  import matplotlib.pyplot as plt
  from catboost import CatBoostClassifier, CatBoostRegressor
  import numpy as np

  # Load data
  train = pd.read_csv('data/german_train_pipelineB.csv')
  val = pd.read_csv('data/german_val_pipelineB.csv')

  # Identify categorical features (object or category types)
  cat_features = [col for col in train.columns if train[col].dtype == 'object' or train[col].dtype == 'category']

  train_y = train['target_good']
  train_X = train.drop(columns=['target_good'])
  val_y = val['target_good']
  val_X = val.drop(columns=['target_good'])

  # Calculate class weights for the Classifier
  counts = train_y.value_counts().to_dict()
  total_train = len(train_y)
  weights = {0: total_train / (2 * counts[0]), 1: total_train / (2 * counts[1])}

  # Train CatBoost Classifier
  cb_classifier = CatBoostClassifier(
      iterations=1000,
      learning_rate=0.05,
      depth=6,
      random_seed=42,
      loss_function='Logloss',
      eval_metric='AUC',
      early_stopping_rounds=100,
      use_best_model=True,
      class_weights=weights
  )
  cb_classifier.fit(train_X, train_y, eval_set=(val_X, val_y), cat_features=cat_features, verbose=False)

  # Train CatBoost Regressor
  cb_regressor = CatBoostRegressor(
      iterations=1000,
      learning_rate=0.05,
      depth=6,
      random_seed=42,
      loss_function='RMSE',
      eval_metric='RMSE',
      use_best_model=True
  )
  cb_regressor.fit(train_X, train_y, eval_set=(val_X, val_y), cat_features=cat_features, verbose=False)

  # Extract results from both models
  res_c = cb_classifier.get_evals_result()
  res_r = cb_regressor.get_eval_result()

  train_logloss = []
  val_logloss = []
  train_rmse = []
  val_rmse = []

  for i in range(len(res_c)):
      # Extract Logloss from Classifier results
      train_logloss.append(res_c[i].get('loss', 0))
      val_logloss.append(res_c[i].get('eval_loss', res_c[i].get('loss', 0)))
      # Extract RMSE from Regressor results
      train_rmse.append(res_r[i].get('loss', 0))
      val_rmse.append(res_r[i].get('eval_loss', res_r[i].get('loss', 0)))

  # Plot the curves
  plt.figure(figsize=(12, 8))
  plt.plot(train_logloss, label='Train Logloss')
  plt.plot(val_logloss, label='Val Logloss')
  plt.plot(train_rmse, label='Train RMSE')
  plt.plot(val_rmse, label='Val RMSE')

  plt.title('CatBoost Training Metrics (Logloss & RMSE)')
  plt.xlabel('Iteration')
  plt.ylabel('Value')
  plt.legend()
  plt.grid(True)
  plt.savefig('training_curves.png')
  print("Plot successfully saved to training_curves.png")
```

```python
  import pandas as pd
  from catboost import CatBoostClassifier, CatBoostRegressor
  import matplotlib.pyplot as plt
  import numpy as np

  # Load and prepare data
  train = pd.read_csv('data/german_train_pipelineB.csv')
  val = pd.read_csv('data/german_val_pipelineB.csv')
  cat_features = [col for col in train.columns if train[col].dtype == 'object' or train[col].dtype == 'category']

  train_y = train['target_good']
  train_X = train.drop(columns=['target_good'])
  val_y = val['target_good']
  val_X = val.drop(columns=['target_good'])

  # Calculate class weights for the classifier
  counts = train_y.value_counts().to_dict()
  total_train = len(train_y)
  weights = {0: total_train / (2 * counts[0]), 1: total_train / (2 * counts[1])}

  # Train models
  cb_classifier = CatBoostClassifier(
      iterations=1000, learning_rate=0.05, depth=6, random_seed=42,
      loss_function='Logloss', eval_metric='AUC', early_stopping_rounds=100,
      use_best_model=True, class_weights=weights
  )
  cb_classifier.fit(train_X, train_y, eval_set=(val_X, val_y), cat_features=cat_features, verbose=False)

  cb_regressor = CatBoostRegressor(
      iterations=1000, learning_rate=0.05, depth=6, random_seed=42,
      loss_function='RMSE', eval_metric='RMSE', use_best_model=True
  )
  cb_regressor.fit(train_X, train_y, eval_set=(val_X, val_y), cat_features=cat_features, verbose=False)

  # Get predictions
  c_probs = cb_classifier.predict_proba(val_X)[:, 1]
  r_preds = cb_regressor.predict(val_X)
  r_probs = np.clip(r_preds, 0, 1)

  # Plotting
  fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(16, 7), sharey=True)
  for label, target in zip(['Good', 'Bad'], [0, 1]):
      mask = (val_y == target)
      ax1.hist(c_probs[mask], bins=20, alpha=0.5, label=label, color='blue' if label=='Good' else 'red')
  ax1.set_title('CatBoost Classifier Predictions')
  ax1.set_xlabel('Predicted Probability')
  ax1.set_ylabel('Frequency')
  ax1.legend()

  for label, target in zip(['Good', 'Bad'], [0, 1]):

      ax2.hist(r_probs[mask], bins=20, alpha=0.5, label=label, color='blue' if label=='Good' else 'red')

  ax2.legend()

  plt.suptitle('Prediction Distributions by Target Class')
  plt.tight_layout()
  plt.savefig('predicted_probability_distributions.png')
```
